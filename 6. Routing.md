## Routing
- Routing is the process of handling different URLs and HTTP requests in a Node.js application.
- It decides which function should execute based on the requested URL and HTTP method.

### HTTP Methods in Express
| Method | Express Syntax | Purpose              |
| ------ | -------------- | -------------------- |
| GET    | `app.get()`    | Read data            |
| POST   | `app.post()`   | Create data          |
| PUT    | `app.put()`    | Update complete data |
| PATCH  | `app.patch()`  | Update partial data  |
| DELETE | `app.delete()` | Delete data          |

### routes/studentRoutes.js

```

import router from "express"

export const router = express.Router();

// Temporary student data
let students = [
    { id: 1, name: "Arun" },
    { id: 2, name: "Kumar" }
];

// GET - Get all students

router.get("/", (req, res) => {
    res.json(students);
});

// GET - Get student by ID
router.get("/:id", (req, res) => {
    const student = students.find(
        (item) => item.id === Number(req.params.id)
    );

    if (!student) {
        return res.status(404).send("Student not found");
    }

    res.json(student);
});

// POST - Add new student
router.post("/", (req, res) => {
    const newStudent = {
        id: Date.now(),
        name: req.body.name
    };

    students.push(newStudent);

    res.status(201).json(newStudent);
});

// PUT - Update complete student
router.put("/:id", (req, res) => {
    const id = Number(req.params.id);

    const index = students.findIndex(
        (item) => item.id === id
    );

    if (index === -1) {
        return res.status(404).send("Student not found");
    }

    students[index] = {
        id: id,
        name: req.body.name
    };

    res.json(students[index]);
});

// PATCH - Update partial student
router.patch("/:id", (req, res) => {
    const id = Number(req.params.id);

    const student = students.find(
        (item) => item.id === id
    );

    if (!student) {
        return res.status(404).send("Student not found");
    }

    Object.assign(student, req.body);

    res.json(student);
});

// DELETE - Delete student
router.delete("/:id", (req, res) => {
    const id = Number(req.params.id);

    students = students.filter(
        (item) => item.id !== id
    );

    res.send("Student deleted successfully");
});

module.exports = router;

```

const express = require("express");

const app = express();

// Middleware to read JSON request body
app.use(express.json());

// Import Student Router
const studentRoutes = require("./routes/studentRoutes");

// Register Student Router
app.use("/students", studentRoutes);

// Start Server
app.listen(8000, () => {
    console.log("Server running on port 8000");
});
