### What is Input Validation?

- Input Validation is the process of checking user input before processing or saving it.

- It makes sure the user provides correct and required data. It helps prevent invalid data from entering our application.

***Basic syntax:***
```
if (condition) {
    // Valid input
}
else {
    // Invalid input
}
```

***Express example:***
```
app.post("/students", (req, res) => {

    const name = req.body.name;

    if (!name) {
        return res.status(400).send("Name is required");
    }

    res.send("Student Added Successfully");

});
```
***Complete Example***
```

const express = require("express");

const app = express();

// Read JSON request body
app.use(express.json());

// Student Registration API
app.post("/students", (req, res) => {

    const { name, age, email } = req.body;

    // Name Validation
    if (!name || name.trim().length < 3) {
        return res.status(400).json({
            message: "Name must contain at least 3 characters"
        });
    }

    // Age Validation
    if (!Number.isInteger(age) || age < 18 || age > 60) {
        return res.status(400).json({
            message: "Age must be between 18 and 60"
        });
    }

    // Email Validation
    if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
        return res.status(400).json({
            message: "Please enter a valid email"
        });
    }

    // Success Response
    res.status(201).json({
        message: "Student Registered Successfully",
        student: { name, age, email }
    });

});

app.listen(8000, () => {
    console.log("Server running on port 8000");
});

````
